\chapter{Mathematical Reference}

\section{Derivative Rules}
\begin{align}
\dv{x} f(g(x)) &= f'(g(x))g'(x) \\
\dv{x}(fg) &= f'g+fg' \\
\dv{x}\left(\frac fg\right) &= \frac{f'g-fg'}{g^2}
\end{align}

\section{Common Derivatives}
\begin{align}
\dv{x}x^n &= nx^{n-1} & \dv{x}\e^x &= \e^x \\
\dv{x}\ln x &= \frac1x & \dv{x}\sin x &= \cos x
\end{align}

\section{Integral Rules}
\begin{align}
\int_a^b f'(x)\,\dd x &= f(b)-f(a) \\
\int u\,\dd v &= uv-\int v\,\dd u
\end{align}

\section{Common Integrals}
\begin{align}
\int x^n\,\dd x &= \frac{x^{n+1}}{n+1}+C \quad(n\neq-1)\\
\int \frac{\dd x}{x} &= \ln|x|+C\\
\int \e^{ax}\,\dd x &= \frac{\e^{ax}}a+C\quad(a\neq0)
\end{align}
