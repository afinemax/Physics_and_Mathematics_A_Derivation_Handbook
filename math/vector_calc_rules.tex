
\section{Vector Calculus Rules}
\label{sec:vector_calc_rules}

\subsection{Triple Products}
\label{subsec:vector_calc_rules_tripple_product}

\begin{equation}
    \mathbf{A}\cdot(\mathbf{B}\times\mathbf{C})
    = \mathbf{B}\cdot(\mathbf{C}\times\mathbf{A})
    = \mathbf{C}\cdot(\mathbf{A}\times\mathbf{B})
    \label{eq:vector_calc_rules_tripple_product_scalar}
\end{equation}
Note that the ABC cyclic order is maintained, this is the scaler tripler product. 

\begin{equation}
    \mathbf{A}\times(\mathbf{B}\times\mathbf{C})
    = \mathbf{B}(\mathbf{A}\cdot\mathbf{C})
    - \mathbf{C}(\mathbf{A}\cdot\mathbf{B})
    \label{eq:vector_calc_rules_tripple_product_vector}
\end{equation}
This is the ``BAC-CAB'' rule, the vector triple product~\eqref{eq:vector_calc_rules_tripple_product_vector}.
